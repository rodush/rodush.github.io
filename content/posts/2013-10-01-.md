I was never thinking to deep about how timeouts work in JavaScript and what will happen if we call some function after 0 milliseconds. Havel you?

Let&#8217;s examine such interesting piece of code (here $ is a jQuery instance):

<pre lang="javascript">var f = function(){
    var d = $.Deferred();
    setTimeout(function() {
        console.log(1);
    }, 0);
    d.promise().then(function() {
        console.log("Expect after");
        setTimeout(function() {
            console.log("Inside promise");
        }, 0);
    });
    setTimeout(function() {
        console.log(2);
    }, 0);
    console.log("Expect before");
    setTimeout(function() {
        console.log(3);
    }, 0);
    d.resolve();
    console.log("After resolve");
}
setTimeout(function(){
    console.log("Out of F() scope");
}, 0);
f();
</pre>

What is the output would you expect? 😉
